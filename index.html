SAMPLE RISK SEGMENTATION SQL QUERY
SELECT c.employment_type,
       AVG(c.credit_score) AS avg_score,
       SUM(l.loan_amount)  AS total_disbursed,
       COUNT(CASE WHEN r.dpd > 30 THEN 1 END) * 100.0 / COUNT(*) AS delinquency_rate_pct
FROM customer c JOIN loan_application l ON c.customer_id = l.customer_id
JOIN repayment r ON l.application_id = r.application_id
GROUP BY c.employment_type;
